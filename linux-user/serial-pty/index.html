01目标

学完这一课,你应当能在没有 root 权限、没有开发板的普通 Linux 上完成下面这些可验证的动作:

  • 用管道、重定向和 chmod 完成一次“文件 → 过滤 → 计数”,并说清 >、>>、2>&1、| 的差别。
  • 用 fork/exec 模型解释进程如何创建;用 ps -o pid,nlwp,stat,comm 区分进程与线程;用 SIGTERM 结束自己启动的进程,并解释退出码为什么是 143。

03原理与图示

3.1 Shell:管道、重定向、权限

Shell 本身也是用户态程序。它把多个小程序的输入输出接起来:| 把左边程序的 stdout 接到右边程序的 stdin; > 覆盖写文件,>> 追加;< 把文件接到 stdin;2>&1 把 stderr 也并到 stdout。 Linux 的权限分三组(属主 / 属组 / 其他人)各有读 r、写 w、执行 x;数字表示法里 644 就是 rw-r--r--。 最小权限原则:只给需要的人必要的位,不要用 chmod 777 图省事。

shell 实测:管道 / 重定向 / 权限 实测命令Ubuntu 24.04 
$ whoami
czy
$ cat sensor.log | wc -l
3
$ wc -c < sensor.log
63
$ ls -l sensor.log
-rw-r--r-- 1 czy czy 63 Nov 27 17:51 sensor.log
$ chmod 600 sensor.log && ls -l sensor.log
-rw------- 1 czy czy 63 Nov 27 17:51 sensor.log
$ cat secret.log
cat: secret.log: Permission denied
exit=1

输出里的用户名、大小、时间都来自本机这次会话;换一台机器会不同。secret.log 是实验临时文件, 用 chmod 000 演示“没有读权限”;演示后立刻恢复权限并删除,不要对系统文件这样做。

3.2 进程、线程与信号

进程是资源分配的单位:它有独立的虚拟地址空间、文件描述符表、信号处理设置。 线程是 CPU 调度的单位:同一进程内的线程共享地址空间和文件描述符,但各自有栈和寄存器上下文。 fork() 复制出一个几乎一样的子进程(写时复制,COW),exec() 则用新程序替换当前进程的代码与数据; 典型的“启动新程序”是 fork 之后在子进程里 exec。

查看进程常用 ps:ps -o pid,ppid,stat,comm 看父子关系与状态,ps -o nlwp 看线程数(大于 1 说明是多线程), ps -T -p <pid> 列出线程。状态列常见值:R 运行、S 可中断睡眠、D 不可中断睡眠、 Z 僵尸、T 停止。结束进程优先发 SIGTERM(15),让程序有机会清理;程序不响应时才考虑 SIGKILL(9)。

3.3 文件 I/O:open / read / write

用户态通过系统调用访问文件。每个打开的文件对应一个文件描述符(小整数,从 0 开始分配;0/1/2 默认是 stdin/stdout/stderr)。 open() 返回 fd,失败返回 -1 并设置 errno;read() 返回实际读到的字节数, 0 表示 EOF,-1 表示出错;write() 可能只写一部分,正确写法要循环到全部写完;close() 释放 fd。 O_CREAT 的权限参数会被 umask 掩掉一部分,通常 0644 得到 rw-r--r--。

下面的程序用 open/write 写 3 帧模拟数据,再用 open/read 循环读回,统计字节数和行数。 3 帧 × 21 字节(20 字符 + 换行)= 63 字节,正好可以和 wc -c 对照。

file_io.c compiled · gcc 13.3.0 
/* file_io.c — 用 open/read/write 写 3 帧模拟数据,再读回统计字节数与行数
 * 编译: gcc -Wall -Wextra -O2 -o file_io file_io.c
 * 运行: ./file_io sensor.log
 * 说明: 数据是模拟传感器帧,不是真实设备采集。
 */
#include <errno.h>
#include <fcntl.h>
#include <stdio.h>
#include <string.h>
#include <unistd.h>

static const char *frames[] = {
    "T=24.5,H=58.2,P=1013\n",
    "T=24.7,H=58.0,P=1013\n",
    "T=24.9,H=57.8,P=1012\n",
};
static const int nframes = (int)(sizeof(frames) / sizeof(frames[0]));

int main(int argc, char **argv)
{
    const char *path = (argc > 1) ? argv[1] : "sensor.log";

    /* 1) open + write: 截断写模式,权限 0644 */
    int fd = open(path, O_WRONLY | O_CREAT | O_TRUNC, 0644);
    if (fd < 0) {
        fprintf(stderr, "open(%s, O_WRONLY): %s\n", path, strerror(errno));
        return 1;
    }
    for (int i = 0; i < nframes; i++) {
        size_t len = strlen(frames[i]);
        ssize_t n = write(fd, frames[i], len);
        if (n != (ssize_t)len) {
            fprintf(stderr, "write frame %d: %s\n", i, strerror(errno));
            close(fd);
            return 1;
        }
    }
    close(fd);
    printf("[write] %d 帧 -> %s\n", nframes, path);

    /* 2) open + read: 循环读直到 read 返回 0(EOF) */
    fd = open(path, O_RDONLY);
    if (fd < 0) {
        fprintf(stderr, "open(%s, O_RDONLY): %s\n", path, strerror(errno));
        return 1;
    }
    char buf[128];
    long bytes = 0, lines = 0;
    ssize_t n;
    while ((n = read(fd, buf, sizeof buf)) > 0) {
        bytes += (long)n;
        for (ssize_t i = 0; i < n; i++)
            if (buf[i] == '\n')
                lines++;
    }
    if (n < 0) {
        fprintf(stderr, "read: %s\n", strerror(errno));
        close(fd);
        return 1;
    }
    close(fd);

    printf("[read ] path=%s bytes=%ld lines=%ld\n", path, bytes, lines);
    printf("[ok   ] 与 wc -c / wc -l 对照\n");
    return 0;
}

3.4 串口与 termios(115200 8N1)

在 Linux 上,串口被抽象成 tty 设备文件:/dev/ttyUSB0(USB 转串口)、/dev/ttyACM0(USB CDC)、 /dev/ttyS0(板载 UART)。打开时常用 O_RDWR | O_NOCTTY,O_NOCTTY 防止这个 tty 变成进程的控制终端。 配置参数放在 struct termios 里,通过 tcgetattr() 读出、修改、tcsetattr() 写回。

“115200 8N1”是四个参数:波特率 115200、8 个数据位、无校验(no parity)、1 个停止位。对应到 termios: cfsetispeed/cfsetospeed(B115200),c_cflag |= CS8,c_cflag &= ~PARENB, c_cflag &= ~CSTOPB。注意 CSTOPB 置位表示 2 个停止位,清掉才是 1 个。 再加 CLOCAL | CREAD 忽略调制解调器控制线并启用接收器。cfmakeraw() 关闭行缓冲、回显、信号字符和 CR/LF 转换, 这是二进制/帧协议场景的常用起点。

termios 设置本课取值作用
cfsetispeed/cfsetospeedB115200输入/输出波特率 115200
c_cflag &= ~CSIZE; c_cflag |= CS8CS88 个数据位
c_cflag &= ~PARENB无校验关闭奇偶校验
c_cflag &= ~CSTOPB1 停止位CSTOPB 置位才是 2 个停止位
c_cflag |= CLOCAL | CREAD忽略 Modem 线、启用接收不依赖 DCD/DSR,允许 read
cfmakeraw()raw 模式关闭行规程、回显、CR/LF 转换与信号字符
VMIN=0, VTIME=100 字节 / 1.0 秒read 最多等 1 秒;返回 0 表示超时或对端关闭
tcsetattr(fd, TCSANOW, &tio)立即生效写回配置;之后用 tcgetattr 回读确认

下面程序打开 tty、配置 115200 8N1、回读确认,然后按行解析 T=..,H=..,P=.. 帧。 本课没有真实串口硬件:运行时由 Python 脚本创建伪终端(PTY),把模拟数据写到 master 端,C 程序从 slave 端读。 termios、tty 行规程和 read 走的是同一套内核代码路径,但数据没有经过 UART 电气层,也没有真实传感器的单位与误差。

serial_termios.c compiled · gcc 13.3.0simulated 数据 
/* serial_termios.c — 打开 tty,配置 115200 8N1 raw 模式,读取模拟传感器帧
 * 编译: gcc -Wall -Wextra -O2 -o serial_termios serial_termios.c
 * 运行: ./serial_termios /dev/ttyUSB0        # 真实串口(需硬件)
 *       python3 sim_serial.py --pty ./serial_termios   # 伪终端模拟(本课)
 * 说明: 页面演示使用 PTY 模拟数据;未在 USB-UART 硬件上验证。
 */
#define _POSIX_C_SOURCE 200809L
#define _DEFAULT_SOURCE
#include <errno.h>
#include <fcntl.h>
#include <stdio.h>
#include <string.h>
#include <termios.h>
#include <unistd.h>

static int serial_config(int fd, const char *dev)
{
    struct termios tio;
    if (tcgetattr(fd, &tio) < 0) {
        perror("tcgetattr");
        return -1;
    }

    cfmakeraw(&tio);                              /* 关行缓冲/回显/CR 转换 */
    tio.c_cflag &= ~(PARENB | CSTOPB | CSIZE | CRTSCTS);
    tio.c_cflag |= CS8 | CLOCAL | CREAD;          /* 8 数据位,无校验,1 停止位 */
    tio.c_iflag &= ~(IXON | IXOFF | IXANY);       /* 关软件流控 */
    tio.c_cc[VMIN] = 0;
    tio.c_cc[VTIME] = 10;                         /* 0.1s 单位:1.0s 读超时 */
    cfsetispeed(&tio, B115200);
    cfsetospeed(&tio, B115200);
    if (tcsetattr(fd, TCSANOW, &tio) < 0) {
        perror("tcsetattr");
        return -1;
    }

    /* 回读确认内核真的接受了设置 */
    struct termios chk;
    if (tcgetattr(fd, &chk) < 0) {
        perror("tcgetattr(chk)");
        return -1;
    }
    int is_8n1 = ((chk.c_cflag & CSIZE) == CS8)
              && !(chk.c_cflag & PARENB)
              && !(chk.c_cflag & CSTOPB);
    int baud_ok = (cfgetispeed(&chk) == B115200) && (cfgetospeed(&chk) == B115200);
    printf("[serial] dev=%s baud=%s 8N1=%s VMIN=%d VTIME=%d\n",
           dev, baud_ok ? "115200" : "mismatch", is_8n1 ? "yes" : "no",
           (int)chk.c_cc[VMIN], (int)chk.c_cc[VTIME]);
    return 0;
}

int main(int argc, char **argv)
{
    const char *dev = (argc > 1) ? argv[1] : "/dev/ttyUSB0";
    int fd = open(dev, O_RDWR | O_NOCTTY | O_NONBLOCK);
    if (fd < 0) {
        fprintf(stderr, "open(%s): %s\n", dev, strerror(errno));
        return 1;
    }
    if (serial_config(fd, dev) < 0) {
        close(fd);
        return 1;
    }

    /* O_NONBLOCK 只用于 open 阶段,读之前恢复阻塞 + VTIME 超时语义 */
    int flags = fcntl(fd, F_GETFL);
    if (flags >= 0)
        fcntl(fd, F_SETFL, flags & ~O_NONBLOCK);

    char line[128];
    size_t used = 0;
    long frames = 0, bytes = 0, bad = 0;
    unsigned long checksum = 0;
    for (;;) {
        char ch;
        ssize_t n = read(fd, &ch, 1);
        if (n == 0)
            break;                    /* VTIME 超时或对端关闭 */
        if (n < 0) {
            if (errno == EINTR)
                continue;
            break;
        }
        bytes++;
        checksum = (checksum + (unsigned char)ch) & 0xFF;
        if (ch == '\n') {
            line[used] = '\0';
            double t = 0, h = 0;
            int p = 0;
            if (used > 0 && sscanf(line, "T=%lf,H=%lf,P=%d", &t, &h, &p) == 3) {
                frames++;
                printf("[frame %ld] T=%.1f H=%.1f P=%d\n", frames, t, h, p);
            } else if (used > 0) {
                bad++;
                printf("[bad   ] %s\n", line);
            }
            used = 0;
        } else if (used + 1 < sizeof line) {
            line[used++] = ch;
        }
    }
    close(fd);
    printf("[serial] frames=%ld bytes=%ld bad=%ld checksum=0x%02lx\n",
           frames, bytes, bad, checksum);
    return (frames > 0) ? 0 : 2;
}
simulated 边界:下面的串口数据由 sim_serial.py 用固定随机种子生成,经 PTY 送入 C 程序。 它能证明 termios 配置、按行解析、字节统计和超时逻辑可运行;不能证明真实 UART 的波特率精度、电气兼容性、 传感器量程或误差。要标 hardware,需要接上具体型号的 USB-TTL/开发板并记录实测条件。

3.5 定时与日志

定时任务不要依赖“睡固定时间再干活”的朴素循环:每次处理耗时都会累计成漂移。正确做法是用单调时钟 CLOCK_MONOTONIC 计算周期,生产代码常用 clock_nanosleep(CLOCK_MONOTONIC, TIMER_ABSTIME, ...) 指定绝对唤醒时刻。给日志打时间戳则要用墙上时钟 CLOCK_REALTIME,因为人要看“几点几分”。 两者分工:单调时钟管间隔,墙上时钟管记录。

日志用 O_APPEND 打开,保证每次 write 追加在文件末尾;多进程写同一文件时,单次 write 的原子性依赖于文件系统与写入方式。生产环境还要考虑轮转(rotating)、落盘时机(fsync 与性能的取舍)以及日志级别。 下面示例每 200 ms 写一行,连跑两次可以验证 O_APPEND 不会覆盖旧内容。

timer_log.c compiled · gcc 13.3.0 
/* timer_log.c — 用 CLOCK_MONOTONIC 控周期、CLOCK_REALTIME 打时间戳,定时写日志
 * 编译: gcc -Wall -Wextra -O2 -o timer_log timer_log.c
 * 运行: ./timer_log timer.log
 * 说明: 只演示周期与追加写;生产代码应使用 clock_nanosleep(TIMER_ABSTIME) 减少累计漂移。
 */
#define _POSIX_C_SOURCE 200809L
#include <errno.h>
#include <fcntl.h>
#include <stdio.h>
#include <string.h>
#include <time.h>
#include <unistd.h>

static void timestamp(char *out, size_t cap)
{
    struct timespec rt;
    clock_gettime(CLOCK_REALTIME, &rt);
    struct tm tm;
    localtime_r(&rt.tv_sec, &tm);
    char base[32];
    strftime(base, sizeof base, "%Y-%m-%d %H:%M:%S", &tm);
    int ms = (int)(rt.tv_nsec / 1000000L);
    snprintf(out, cap, "%s.%03d", base, ms);
}

int main(int argc, char **argv)
{
    const char *path = (argc > 1) ? argv[1] : "timer.log";
    int fd = open(path, O_WRONLY | O_CREAT | O_APPEND, 0644);
    if (fd < 0) {
        fprintf(stderr, "open(%s): %s\n", path, strerror(errno));
        return 1;
    }

    struct timespec t0;
    clock_gettime(CLOCK_MONOTONIC, &t0);
    for (int i = 0; i < 4; i++) {
        struct timespec now, req;
        clock_gettime(CLOCK_MONOTONIC, &now);
        double elapsed = (double)(now.tv_sec - t0.tv_sec)
                       + (double)(now.tv_nsec - t0.tv_nsec) / 1e9;
        char ts[40], line[128];
        timestamp(ts, sizeof ts);
        int len = snprintf(line, sizeof line,
                           "[%s] seq=%d elapsed=%.1fs\n", ts, i, elapsed);
        if (write(fd, line, (size_t)len) != len) {
            fprintf(stderr, "write: %s\n", strerror(errno));
            close(fd);
            return 1;
        }
        printf("%s", line);
        req.tv_sec = 0;
        req.tv_nsec = 200L * 1000L * 1000L; /* 200 ms */
        nanosleep(&req, NULL);
    }
    close(fd);
    return 0;
}

#!/usr/bin/env python3
"""sim_serial.py — 生成模拟串口数据;可选通过 PTY 喂给 termios 程序。

注意:数据由随机数生成,是教学模拟,不是真实串口/传感器采集。
不要据此声称硬件验证;真实串口请用 /dev/ttyUSB0 等设备路径。
"""
import argparse
import os
import pty
import random
import subprocess
import sys
import time


def make_frames(count, seed):
    rng = random.Random(seed)
    temp, hum, pres = 24.5, 58.0, 1013
    frames = []
    for _ in range(count):
        temp += rng.uniform(-0.3, 0.3)
        hum += rng.uniform(-0.4, 0.4)
        pres += rng.choice((-1, 0, 0, 1))
        frames.append(f"T={temp:.1f},H={hum:.1f},P={pres}\n")
    return frames


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--count", type=int, default=6)
    ap.add_argument("--seed", type=int, default=7)
    ap.add_argument("--out", default="sim_serial.log")
    ap.add_argument("--pty", metavar="PROG",
                    help="把数据经伪终端喂给该程序(如 ./serial_termios)")
    args = ap.parse_args()

    frames = make_frames(args.count, args.seed)
    with open(args.out, "w", encoding="utf-8") as f:
        f.writelines(frames)
    print(f"[sim] frames={len(frames)} seed={args.seed} -> {args.out}", flush=True)
    for i, line in enumerate(frames, 1):
        print(f"[sim] tx {i}: {line.rstrip()}", flush=True)

    if not args.pty:
        return 0

    master, slave = pty.openpty()
    slave_name = os.ttyname(slave)
    print(f"[pty] slave={slave_name} (伪终端,不是硬件串口)", flush=True)
    proc = subprocess.Popen([args.pty, slave_name])
    time.sleep(0.4)  # 等 C 程序完成 tcsetattr
    for line in frames:
        os.write(master, line.encode())
        time.sleep(0.1)
    time.sleep(0.2)
    os.close(master)
    os.close(slave)
    rc = proc.wait(timeout=5)
    print(f"[pty] 子进程退出码={rc}", flush=True)
    return rc


if __name__ == "__main__":
    sys.exit(main())

06故障排查

每条按“现象 → 检查步骤 → 预期结果 → 下一分支”组织。先复现、再定位,不要一上来就改代码或加 sudo。

1. 权限不足:能打开设备/文件,但读不了或写不了
现象
cat: secret.log: Permission denied、open(...): Permission denied,或非 root 用户访问 /dev/ttyUSB0 报错。
检查步骤
ls -l 文件或设备 看权限位与属主;id 看自己属于哪些组;namei -l 路径 逐级检查目录权限;串口设备再看 ls -l /dev/ttyUSB0 的组是不是 dialout。
预期结果
文件权限为 rw-r--r-- 且属主是你;串口设备组为 dialout(Ubuntu/Debian),而 id 的输出里包含该组。/dev 下的设备通常只有 root 和对应组可读写。
下一分支
文件只给自己读写:chmod 600 文件(不要 chmod 777)。串口:sudo usermod -aG dialout $USER,重新登录(组变更不会影响已登录会话)后再试;或使用 udev 规则固定权限。仍失败时检查是否被 SELinux/AppArmor 或容器权限限制。

2. 串口设备名变化:/dev/ttyUSB0 时有时无
现象
昨天程序用 /dev/ttyUSB0 正常,今天变成 /dev/ttyUSB1 或 /dev/ttyACM0;插拔多个 USB 设备后编号漂移。
检查步骤
ls -l /dev/ttyUSB* /dev/ttyACM* 看当前节点;插拔前后各跑一次 dmesg | tail -n 20 看内核识别日志;用 udevadm info -q property -n /dev/ttyUSB0 查 ID_SERIAL、ID_VENDOR_ID、ID_MODEL_ID;检查 /dev/serial/by-id/ 是否有稳定软链接。
预期结果
设备节点与 USB 设备的序列号/VID:PID 一一对应;/dev/serial/by-id/usb-...-if00-port0 这类路径在重新插拔后保持不变。
下一分支
程序不要硬编码 ttyUSB0,把设备路径作为命令行参数;优先使用 /dev/serial/by-id/ 下的稳定路径;需要固定名字/权限时写 udev 规则(SYMLINK+=、MODE=、GROUP=)。
3. 波特率或帧格式不匹配:读到乱码、坏帧、超时

预期结果
stty -a 显示 speed 115200 baud,且标志包含 cs8、-parenb、-cstopb;发送已知字符串能原样收到,帧格式与代码一致。
下一分支

4. 进程僵尸:ps 的 STAT 列显示 Z,kill 也杀不掉
现象

5. 读完数据不返回/只读到半帧:VMIN、VTIME 与部分读
现象

检查步骤
检查 c_cc[VMIN] 与 c_cc[VTIME] 的设置;打印每次 read 的返回值;检查是否处理了 errno == EINTR;确认发送端是否按行发送、行尾是 \n 还是 \r\n。
预期结果
VMIN=0, VTIME=10 时 read 最多等 1 秒,超时返回 0;正常收到数据时返回实际字节数。按行缓冲并处理半帧后,每帧应完整解析。
下一分支
不要假设一次 read 等于一帧:维护行缓冲,按 \n 切分,处理跨 read 的半帧;对 EINTR 重试;上位机与设备端统一行尾;必要时加帧头、长度和校验和做帧同步。

1. 同一进程内的两个线程,哪些资源是共享的?
正确答案:A。线程是调度单位,共享进程的地址空间、fd 表、信号处理等;每个线程只独享栈和寄存器上下文。进程之间才有各自独立的地址空间。实验里 ps -o nlwp 的 NLWP=2 就是同一进程有两个线程。
2. 要把串口配成 115200 8N1,下面哪组 termios 设置正确?
正确答案:A。8N1 = 8 数据位(CS8)、无校验(清 PARENB)、1 停止位(清 CSTOPB);CSTOPB 置位反而是 2 个停止位。配置完用 tcgetattr 回读确认,并可再用 stty -F 设备 -a 核对 speed 115200 与 cs8 -parenb -cstopb。
3. read(fd, buf, sizeof buf) 返回 0,表示什么?
正确答案:A。read 返回值:大于 0 是实际读到的字节数;等于 0 是 EOF(对 tty 也可能是 VTIME 超时/对端关闭);等于 -1 才是错误,需要看 errno;被信号打断时通常是 -1 且 errno == EINTR,应重试。
4. 子进程已经退出,父进程没有调用 waitpid,ps 里 STAT 显示 Z。正确处理是?
正确答案:A。僵尸进程已经死了,只剩一个退出状态等待父进程读取;kill 已无进程可杀。父进程应调用 wait()/waitpid()(可在 SIGCHLD 处理里循环 WNOHANG),或让父进程退出由 init/systemd 收养回收。重启系统不是正确做法。
