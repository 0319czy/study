方向目标:用一个能跑的最小机器人把 ROS 2 的三条主线串起来——话题传数据、TF 统一坐标、Nav2 把目标变成速度。 计算图要能点开看、能解释每条边的方向与 QoS,并用日志和 rosbag 证明它真的动了。

3. 原理与图示

ROS 2 不是"一个库",而是一套约定:节点通过命名的 topic / service / action 通信,坐标系通过 TF 统一,生命周期节点由状态机管理。先学会读图,再写代码。

3.1 四种通信方式

机制方向与语义典型用途关键点
话题 topic多对多、异步、有队列/scan、/odom、/cmd_vel、/tf发布者不关心谁在听;QoS 必须兼容
服务 service一对一、请求/响应、可超时保存地图、清空代价地图、查询状态不要用它传持续数据流
动作 action长任务 + 反馈 + 可取消/navigate_to_pose、机械臂抓取goal / feedback / result 三段式
参数 parameter节点内配置、可运行时修改速度上限、代价地图膨胀半径改完要确认节点是否真的更新

3.2 DDS 发现与 QoS:为什么"话题在,数据没有"

ROS 2 默认用 DDS 做去中心化发现:没有 master,节点靠组播/单播互相认识。ROS_DOMAIN_ID 不同、网络不通、容器没共享网络,都会表现为"两个终端互相看不见"。话题存在但收不到,则优先查 QoS。

发布者 \ 订阅者订阅 reliable订阅 best_effort
发布 reliable兼容兼容(订阅端接受较弱保证)
发布 best_effort不兼容兼容

durability 同理:订阅端要 transient_local(晚加入也能拿到最后一帧),发布端必须是 transient_local;发布端 volatile + 订阅端 transient_local 直接不兼容。传感器数据常配 best_effort + volatile;地图 /map 与 /tf_static 常配 transient_local。

sensor_data  : best_effort / volatile       → 允许丢帧,低延迟
map          : reliable / transient_local    → 晚加入的订阅者也能拿到整张图
tf_static    : reliable / transient_local    → 静态变换只发一次
tf (dynamic) : reliable / volatile           → 高频、带时间戳的变换流

3.3 TF 树:一个子帧只能有一个父帧

map ──(定位或建图:SLAM / AMCL)──▶ odom ──(底盘驱动:轮速里程计)──▶ base_link
                                                                  │
                                                     (URDF / static) ├──▶ laser
                                                                     └──▶ camera_link
变换通常由谁发布作用断掉会怎样
map → odomSLAM 建图节点或 AMCL 定位把地图坐标对齐到里程计坐标,吸收累积误差导航/建图报 "target frame does not exist" 或地图漂移
odom → base_link底盘驱动(轮速积分)机器人相对起点在哪里costmap 无法把雷达点投到全局,规划失败
base_link → laserURDF + robot_state_publisher / static_transform_publisher雷达安装外参障碍物位置整体偏移,可能"看不见"墙

建图时由 SLAM 发布 map→odom;切到导航时由 AMCL 接替。两个节点同时发布同一段 TF 会互相打架——切换前先停掉建图节点。查链:ros2 run tf2_ros tf2_echo map laser;画树:ros2 run tf2_tools view_frames。

3.4 Nav2:从目标到 /cmd_vel

RViz 2D Goal / action ──▶ bt_navigator(行为树)
                              │
                              ├──▶ planner_server ──/plan──▶ controller_server ──/cmd_vel──▶ 底盘
                              │         ▲                         ▲
                              │         │                         │
                              └──▶ 全局/局部 costmap ◀── /map、/scan、TF
                                     (static + obstacle + inflation 层)

Nav2 的节点是生命周期节点:unconfigured → inactive → active。lifecycle_manager 负责按顺序 configure/activate;如果 controller_server 停在 inactive,目标可能被拒绝或超时,链路上不会出现 /cmd_vel。用 ros2 lifecycle get /controller_server 确认状态。

3.5 时间戳与执行器:两个隐蔽的坑

  • TF 时间戳:每条变换都带时间。用消息头的时间去查 TF,查不到会报 extrapolation;回放 rosbag 时忘了 use_sim_time:=true 是最常见原因。
  • 执行器:单线程执行器里一个阻塞回调会饿死定时器与订阅;长任务用 MultiThreadedExecutor + callback group,或拆到独立节点。

/* tfcheck.c —— TF 链与四元数检查器(教学用,无 ROS 依赖)
 *
 * 从标准输入读 "parent child" 边,检查:
 *   1) 每个子帧是否只有一个父帧(TF 树不能有多个父);
 *   2) 从目标帧回溯到根帧的链路是否成环;
 *   3) 可选:把 (x,y,z,w) 四元数归一化并换算成 yaw(度)。
 *
 * 编译: gcc -std=c11 -O2 -Wall -Wextra -Werror -o tfcheck tfcheck.c -lm
 * 运行: ./tfcheck laser 0 0 0.7071068 0.7071068 <<'EOF'
 *       map odom
 *       odom base_link
 *       base_link laser
 *       EOF
 * 退出码: 0 通过;1 用法/输入错误;2 多父;3 成环;4 目标帧不存在
 */
#include <math.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_EDGE 32
#define NAME_LEN 32

#ifndef M_PI
#define M_PI 3.14159265358979323846
#endif

struct edge {
  char parent[NAME_LEN];
  char child[NAME_LEN];
};

static struct edge g_edge[MAX_EDGE];
static int g_n;

static int read_edges(void)
{
  char parent[NAME_LEN], child[NAME_LEN];
  while (scanf("%31s %31s", parent, child) == 2) {
    if (g_n >= MAX_EDGE) {
      fprintf(stderr, "边数超过 %d,请拆分 TF 树检查\n", MAX_EDGE);
      return -1;
    }
    snprintf(g_edge[g_n].parent, NAME_LEN, "%s", parent);
    snprintf(g_edge[g_n].child, NAME_LEN, "%s", child);
    g_n++;
  }
  return g_n;
}

/* 返回父边下标;没有父边返回 -1;多个不同父帧返回 -2 */
static int parent_of(const char *child)
{
  int found = -1;
  for (int i = 0; i < g_n; i++) {
    if (strcmp(g_edge[i].child, child) != 0)
      continue;
    if (found >= 0 && strcmp(g_edge[found].parent, g_edge[i].parent) != 0)
      return -2;
    found = i;
  }
  return found;
}

static int frame_exists(const char *name)
{
  for (int i = 0; i < g_n; i++)
    if (strcmp(g_edge[i].parent, name) == 0 || strcmp(g_edge[i].child, name) == 0)
      return 1;
  return 0;
}

int main(int argc, char **argv)
{
  if (argc != 2 && argc != 6) {
    fprintf(stderr, "用法: %s <目标帧> [qx qy qz qw]\n", argv[0]);
    return 1;
  }
  if (read_edges() < 0)
    return 1;
  if (g_n == 0) {
    fprintf(stderr, "标准输入没有 \"parent child\" 边\n");
    return 1;
  }

  const char *target = argv[1];
  if (!frame_exists(target)) {
    fprintf(stderr, "目标帧 %s 不在输入边中(检查 frame_id 拼写)\n", target);
    return 4;
  }

  char chain[MAX_EDGE + 1][NAME_LEN];
  int depth = 0;
  snprintf(chain[depth++], NAME_LEN, "%s", target);

  char current[NAME_LEN];
  snprintf(current, NAME_LEN, "%s", target);
  while (1) {
    int idx = parent_of(current);
    if (idx == -2) {
      fprintf(stderr, "FAIL: 帧 %s 有多个不同父帧,TF 树非法\n", current);
      return 2;
    }
    if (idx < 0)
      break; /* 到根帧 */
    const char *next = g_edge[idx].parent;
    for (int i = 0; i < depth; i++) {
      if (strcmp(chain[i], next) == 0) {
        fprintf(stderr, "FAIL: 检测到环 %s → %s\n", current, next);
        return 3;
      }
    }
    if (depth >= MAX_EDGE) {
      fprintf(stderr, "FAIL: 链路过长,疑似成环\n");
      return 3;
    }
    snprintf(current, NAME_LEN, "%s", next);
    snprintf(chain[depth++], NAME_LEN, "%s", next);
  }

  printf("== TF 链检查 / mybot ==\n");
  printf("边数        : %d\n", g_n);
  printf("目标帧      : %s\n", target);
  printf("根帧        : %s\n", chain[depth - 1]);
  printf("深度        : %d\n", depth - 1);
  printf("链路        : ");
  for (int i = depth - 1; i >= 0; i--)
    printf("%s%s", chain[i], i == 0 ? "\n" : " → ");
  printf("检查        : OK(每个子帧唯一父帧,链路无环)\n");
  printf("节点关系    : ");
  for (int i = 0; i < g_n; i++)
    printf("%s→%s%s", g_edge[i].parent, g_edge[i].child, i == g_n - 1 ? "\n" : "  ");

  if (argc == 6) {
    double qx = atof(argv[2]), qy = atof(argv[3]), qz = atof(argv[4]), qw = atof(argv[5]);
    double norm = sqrt(qx * qx + qy * qy + qz * qz + qw * qw);
    if (norm < 1e-9) {
      fprintf(stderr, "四元数模长为 0,非法\n");
      return 1;
    }
    qx /= norm;
    qy /= norm;
    qz /= norm;
    qw /= norm;
    double yaw = atan2(2.0 * (qw * qz + qx * qy), 1.0 - 2.0 * (qy * qy + qz * qz));
    printf("四元数      : (%.6f, %.6f, %.6f, %.6f) |q| = %.6f\n", qx, qy, qz, qw,
           sqrt(qx * qx + qy * qy + qz * qz + qw * qw));
    printf("yaw         : %.2f deg(绕 z 轴,逆时针为正)\n", yaw * 180.0 / M_PI);
  }
  return 0;
}

说明:该工具在主机上编译运行通过,验证的是 TF 链算法与 yaw 换算;输入是手工写的边表,不是从 ROS 图实时读取。真实系统的 ros2 run tf2_tools view_frames 输出才是运行证据(hardware-pending)。

一个最小底盘节点要同时做三件事:订阅 /cmd_vel 驱动轮子、发布 /odom、广播 odom→base_link。雷达可以用真实驱动或仿真插件发布 /scan。核心 TF 广播如下(节选):

展开 rclpy TF 广播节选(principle:本机未安装 ROS 2,未运行) 
import math
from geometry_msgs.msg import TransformStamped
from tf2_ros import TransformBroadcaster

class BaseNode(Node):                     # 省略 Node 导入与 __init__
    def publish_tf(self, x, y, yaw, stamp):
        t = TransformStamped()
        t.header.stamp = stamp            # 必须用数据时间,不是处理时间
        t.header.frame_id = 'odom'
        t.child_frame_id = 'base_link'
        t.transform.translation.x = x
        t.transform.translation.y = y
        t.transform.rotation.z = math.sin(yaw / 2.0)
        t.transform.rotation.w = math.cos(yaw / 2.0)
        self.tf.sendTransform(t)
